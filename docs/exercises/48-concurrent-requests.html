Exercise 48 · Level 15 — Async Rust

Concurrent Requests

Run many simulated requests concurrently on a single thread with a hand-written join_all, and observe that total time ≈ the slowest request, not the sum.

Advancedcomposing futures`Pin<Box<F>>``Unpin`polling multiple childrenconcurrency vs. parallelism`async fn`

Real projects use a runtime. With Tokio you'd write tokio::join!(a, b) or futures::future::join_all(v).await. This exercise is std-only so you can see how join_all works under the hood. The tiny block_on/sleep runtime from exercise 47 is provided.

Instructions

  1. Implement struct JoinAll<F: Future> { futures: Vec<Option<Pin<Box<F>>>>, outputs: Vec<Option<F::Output>> } and the constructor: 
    fn join_all<I>(futures: I) -> JoinAll<I::Item> where I: IntoIterator, I::Item: Future
  2. Implement Future for JoinAll<F> with Output = Vec<F::Output>: poll every still-pending child; store finished outputs and set their slot to None; return Ready with outputs in the original order once every child is done.
  3. Implement async fn fake_request(id: usize, latency: Duration) -> Response (awaits sleep(latency), returns Response { id, body }).
  4. Implement: - async fn fetch_sequential(latencies: &[u64]) -> Vec<Response> — .await one after another. - async fn fetch_concurrent(latencies: &[u64]) -> Vec<Response> — build all futures, then join_all(..).await.
  5. In main, time both versions with Instant.

Expected behavior

For latencies [80, 30, 100, 50, 60] ms:

sequential: 5 responses in ~320ms (sum = 320ms)
concurrent: 5 responses in ~100ms (max = 100ms)
  • join_all returns outputs in input order, even if later futures finish first.
  • join_all of an empty Vec resolves immediately to vec![].

Optional hints

  • Calling an async fn only creates a future; nothing runs until it's polled. Sequential .awaits therefore add up.
  • Never poll a future again after it returned Ready — that's why completed slots become None.
  • Boxing each child (Box::pin) makes JoinAll Unpin, so poll can use &mut *self freely.

Bonus challenge

The Tokio equivalent (with tokio and futures crates):

#[tokio::main]
async fn main() {
    let (a, b) = tokio::join!(fake_request(0, ms(80)), fake_request(1, ms(30)));
    let all = futures::future::join_all((0..5).map(|i| fake_request(i, ms(50)))).await;
}

use std::pin::{pin, Pin};

// ---------- tiny runtime (see exercise 47) ----------

struct ThreadWaker(Thread);

impl Wake for ThreadWaker {
    fn wake(self: Arc<Self>) {
        self.0.unpark();
    }
}

fn block_on<F: Future>(fut: F) -> F::Output {
    let mut fut = pin!(fut);
    let waker = Waker::from(Arc::new(ThreadWaker(thread::current())));
    let mut cx = Context::from_waker(&waker);
    loop {
        if let Poll::Ready(out) = fut.as_mut().poll(&mut cx) {
            return out;
        }
        thread::park();
    }
}

#[derive(Default)]
struct TimerState {
    completed: bool,
    waker: Option<Waker>,
}

struct TimerFuture {
    state: Arc<Mutex<TimerState>>,
}

fn sleep(duration: Duration) -> TimerFuture {
    let state = Arc::new(Mutex::new(TimerState::default()));
    let thread_state = Arc::clone(&state);
    thread::spawn(move || {
        thread::sleep(duration);
        let mut s = thread_state.lock().unwrap();
        s.completed = true;
        if let Some(w) = s.waker.take() {
            w.wake();
        }
    });
    TimerFuture { state }
}

impl Future for TimerFuture {
    type Output = ();
    fn poll(self: Pin<&mut Self>, cx: &mut Context<'_>) -> Poll<()> {
        let mut s = self.state.lock().unwrap();
        if s.completed {
            Poll::Ready(())
        } else {
            s.waker = Some(cx.waker().clone());
            Poll::Pending
        }
    }
}

// ---------- join_all ----------

/// Polls every child future each time it is woken; resolves once all are done,
/// returning outputs in the original order.
struct JoinAll<F: Future> {
    futures: Vec<Option<Pin<Box<F>>>>,
    outputs: Vec<Option<F::Output>>,
}

fn join_all<I>(futures: I) -> JoinAll<I::Item>
where
    I: IntoIterator,
    I::Item: Future,
{
    let futures: Vec<_> = futures.into_iter().map(|f| Some(Box::pin(f))).collect();
    let outputs = futures.iter().map(|_| None).collect();
    JoinAll { futures, outputs }
}

// Every field is boxed or plain data, so JoinAll is Unpin regardless of F.
impl<F: Future> Unpin for JoinAll<F> {}

impl<F: Future> Future for JoinAll<F> {
    type Output = Vec<F::Output>;

    fn poll(mut self: Pin<&mut Self>, cx: &mut Context<'_>) -> Poll<Self::Output> {
        let this = &mut *self;
        let mut all_done = true;
        for (slot, out) in this.futures.iter_mut().zip(this.outputs.iter_mut()) {
            if let Some(fut) = slot {
                match fut.as_mut().poll(cx) {
                    Poll::Ready(v) => {
                        *out = Some(v);
                        *slot = None; // never poll a completed future again
                    }
                    Poll::Pending => all_done = false,
                }
            }
        }
        if all_done {
            Poll::Ready(this.outputs.iter_mut().map(|o| o.take().unwrap()).collect())
        } else {
            Poll::Pending
        }
    }
}

// ---------- fake requests ----------

#[derive(Debug, PartialEq)]
struct Response {
    id: usize,
    body: String,
}

async fn fake_request(id: usize, latency: Duration) -> Response {
    sleep(latency).await;
    Response {
        id,
        body: format!("payload #{id} after {}ms", latency.as_millis()),
    }
}

async fn fetch_sequential(latencies: &[u64]) -> Vec<Response> {
    let mut out = Vec::new();
    for (id, &ms) in latencies.iter().enumerate() {
        out.push(fake_request(id, Duration::from_millis(ms)).await);
    }
    out
}

async fn fetch_concurrent(latencies: &[u64]) -> Vec<Response> {
    join_all(
        latencies
            .iter()
            .enumerate()
            .map(|(id, &ms)| fake_request(id, Duration::from_millis(ms))),
    )
    .await
}
