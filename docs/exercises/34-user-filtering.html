Exercise 34 · Level 10 — Closures and Iterators

User Filtering

Query a list of users with iterator chains and reusable, composable closure-based filters.

Intermediateiterators over structsborrowing in closures`filter``map``any`/`all``partition``max_by_key``sort_by`grouping with `HashMap`/`BTreeMap`boxed closures (`Box<dyn Fn>`)

Instructions

Given:

#[derive(Debug, Clone, PartialEq)]
enum Role { Admin, Editor, Viewer }

#[derive(Debug, Clone, PartialEq)]
struct User { name: String, age: u32, role: Role, active: bool }

Implement:

  1. active_adults(users: &[User]) -> Vec<&User> — active users aged 18+.
  2. names_with_role(users: &[User], role: &Role) -> Vec<&str> — names sorted alphabetically.
  3. oldest(users: &[User]) -> Option<&User>.
  4. split_by_active(users: &[User]) -> (Vec<&User>, Vec<&User>) — (active, inactive) using partition.
  5. group_by_role(users: &[User]) -> BTreeMap<String, Vec<&str>> — key is format!("{:?}", role).
  6. has_admin(users: &[User]) -> bool and all_named(users: &[User]) -> bool (no empty names).
  7. Composable filters: - type UserFilter = Box<dyn Fn(&User) -> bool>; - min_age(age: u32) -> UserFilter, with_role(role: Role) -> UserFilter, name_contains(needle: &str) -> UserFilter - apply_filters<'a>(users: &'a [User], filters: &[UserFilter]) -> Vec<&'a User> — keep users that match all filters.

Expected behavior

With sample users Ana (34, Admin, active), Ben (17, Viewer, active), Cleo (25, Editor, inactive), Dan (41, Editor, active):

  • active_adults => Ana, Dan
  • names_with_role(&users, &Role::Editor) => ["Cleo", "Dan"]
  • oldest => Dan
  • split_by_active => 3 active, 1 inactive
  • group_by_role => {"Admin": ["Ana"], "Editor": ["Cleo", "Dan"], "Viewer": ["Ben"]}
  • apply_filters(&users, &[min_age(30), with_role(Role::Editor)]) => Dan

Optional hints

  • Return Vec<&User> to avoid cloning; the borrow checker ties the result to users.
  • name_contains must own its needle (needle.to_lowercase()) because the closure outlives the argument.
  • filters.iter().all(|f| f(user)).

Bonus challenge

Add fn and(a: UserFilter, b: UserFilter) -> UserFilter and fn not(f: UserFilter) -> UserFilter combinators.

use std::collections::BTreeMap;

#[derive(Debug, Clone, PartialEq)]
enum Role {
    Admin,
    Editor,
    Viewer,
}

#[derive(Debug, Clone, PartialEq)]
struct User {
    name: String,
    age: u32,
    role: Role,
    active: bool,
}

type UserFilter = Box<dyn Fn(&User) -> bool>;

fn active_adults(users: &[User]) -> Vec<&User> {
    users.iter().filter(|u| u.active && u.age >= 18).collect()
}

fn names_with_role<'a>(users: &'a [User], role: &Role) -> Vec<&'a str> {
    let mut names: Vec<&str> = users
        .iter()
        .filter(|u| &u.role == role)
        .map(|u| u.name.as_str())
        .collect();
    names.sort_unstable();
    names
}

fn oldest(users: &[User]) -> Option<&User> {
    users.iter().max_by_key(|u| u.age)
}

fn split_by_active(users: &[User]) -> (Vec<&User>, Vec<&User>) {
    users.iter().partition(|u| u.active)
}

fn group_by_role(users: &[User]) -> BTreeMap<String, Vec<&str>> {
    users.iter().fold(BTreeMap::new(), |mut groups, u| {
        groups
            .entry(format!("{:?}", u.role))
            .or_insert_with(Vec::new)
            .push(u.name.as_str());
        groups
    })
}

fn has_admin(users: &[User]) -> bool {
    users.iter().any(|u| u.role == Role::Admin)
}

fn all_named(users: &[User]) -> bool {
    users.iter().all(|u| !u.name.trim().is_empty())
}

fn min_age(age: u32) -> UserFilter {
    Box::new(move |u| u.age >= age)
}

fn with_role(role: Role) -> UserFilter {
    Box::new(move |u| u.role == role)
}

fn name_contains(needle: &str) -> UserFilter {
    let needle = needle.to_lowercase();
    Box::new(move |u| u.name.to_lowercase().contains(&needle))
}

fn and(a: UserFilter, b: UserFilter) -> UserFilter {
    Box::new(move |u| a(u) && b(u))
}

fn not(f: UserFilter) -> UserFilter {
    Box::new(move |u| !f(u))
}

fn apply_filters<'a>(users: &'a [User], filters: &[UserFilter]) -> Vec<&'a User> {
    users
        .iter()
        .filter(|u| filters.iter().all(|f| f(u)))
        .collect()
}

fn sample_users() -> Vec<User> {
    let make = |name: &str, age, role, active| User {
        name: name.to_string(),
        age,
        role,
        active,
    };
    vec![
        make("Ana", 34, Role::Admin, true),
        make("Ben", 17, Role::Viewer, true),
        make("Cleo", 25, Role::Editor, false),
        make("Dan", 41, Role::Editor, true),
    ]
}

fn names<'a>(users: &[&'a User]) -> Vec<&'a str> {
    users.iter().map(|u| u.name.as_str()).collect()
}

fn main() {
    let users = sample_users();

    println!("active adults: {:?}", names(&active_adults(&users)));
    println!("editors: {:?}", names_with_role(&users, &Role::Editor));
    println!("oldest: {:?}", oldest(&users).map(|u| &u.name));

    let (active, inactive) = split_by_active(&users);
    println!(
        "active: {:?}, inactive: {:?}",
        names(&active),
        names(&inactive)
    );

    println!("by role: {:?}", group_by_role(&users));
    println!(
        "has admin: {}, all named: {}",
        has_admin(&users),
        all_named(&users)
    );

    let filters = vec![min_age(30), with_role(Role::Editor)];
    println!("30+ editors: {:?}", names(&apply_filters(&users, &filters)));

    let combo = vec![and(name_contains("a"), not(with_role(Role::Admin)))];
    println!(
        "'a' in name, not admin: {:?}",
        names(&apply_filters(&users, &combo))
    );
}
