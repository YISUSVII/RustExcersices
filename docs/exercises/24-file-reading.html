Exercise 24 · Level 8 — Error Handling

Safe File Reading

Read and write files with std::fs, propagate I/O errors with ?, and recover gracefully from a missing file.

Intermediate`std::fs``std::io::Result`the `?` operator`io::ErrorKind::NotFound``std::env::temp_dir`

Instructions

The program is self-contained: it creates its own file under std::env::temp_dir() and deletes it at the end.

  1. Implement write_sample(path: &Path, contents: &str) -> io::Result<()>.
  2. Implement read_lines(path: &Path) -> io::Result<Vec<String>>.
  3. Implement count_non_empty_lines(path: &Path) -> io::Result<usize> (lines that are not just whitespace), reusing read_lines with ?.
  4. Implement read_or_default(path: &Path, default: &str) -> io::Result<String>: - return the file contents if it exists, - return default when the error kind is NotFound, - propagate any other error.
  5. In main, write a sample file, print its lines and non-empty count, try a missing file, then clean up with fs::remove_file.

Expected behavior

wrote <temp dir>/rust-ex24-sample-<pid>.txt
1: first line
2:
3: third line
non-empty lines: 2
missing file -> "(no config found)"
missing file lines -> error: NotFound
cleaned up

Optional hints

  • fs::write, fs::read_to_string, str::lines.
  • match fs::read_to_string(path) { Err(e) if e.kind() == io::ErrorKind::NotFound => ..., other => other }
  • Include std::process::id() in the file name so parallel runs don't collide.

Bonus challenge

Read the file line by line with BufReader::new(File::open(path)?).lines() instead of loading it all at once.

use std::fs;
use std::io::{self, BufRead, BufReader};
use std::path::{Path, PathBuf};

fn write_sample(path: &Path, contents: &str) -> io::Result<()> {
    fs::write(path, contents)
}

fn read_lines(path: &Path) -> io::Result<Vec<String>> {
    let contents = fs::read_to_string(path)?;
    Ok(contents.lines().map(String::from).collect())
}

fn count_non_empty_lines(path: &Path) -> io::Result<usize> {
    let lines = read_lines(path)?;
    Ok(lines.iter().filter(|l| !l.trim().is_empty()).count())
}

fn read_or_default(path: &Path, default: &str) -> io::Result<String> {
    match fs::read_to_string(path) {
        Err(e) if e.kind() == io::ErrorKind::NotFound => Ok(default.to_string()),
        other => other,
    }
}

// Bonus: stream lines instead of loading the whole file.
fn read_lines_buffered(path: &Path) -> io::Result<Vec<String>> {
    let reader = BufReader::new(fs::File::open(path)?);
    reader.lines().collect()
}

fn temp_file(name: &str) -> PathBuf {
    std::env::temp_dir().join(format!("rust-ex24-{name}-{}.txt", std::process::id()))
}

fn run() -> io::Result<()> {
    let path = temp_file("sample");
    write_sample(&path, "first line\n\nthird line\n")?;
    println!("wrote {}", path.display());

    for (i, line) in read_lines(&path)?.iter().enumerate() {
        println!("{}: {line}", i + 1);
    }
    println!("non-empty lines: {}", count_non_empty_lines(&path)?);
    assert_eq!(read_lines_buffered(&path)?, read_lines(&path)?);

    let missing = temp_file("does-not-exist");
    println!(
        "missing file -> {:?}",
        read_or_default(&missing, "(no config found)")?
    );
    match read_lines(&missing) {
        Ok(lines) => println!("unexpected: {lines:?}"),
        Err(e) => println!("missing file lines -> error: {:?}", e.kind()),
    }

    fs::remove_file(&path)?;
    println!("cleaned up");
    Ok(())
}

fn main() {
    if let Err(e) = run() {
        eprintln!("I/O error: {e}");
        std::process::exit(1);
    }
}
