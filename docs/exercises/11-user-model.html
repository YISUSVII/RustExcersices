Exercise 11 · Level 5 — Structs and Enums

User Model

Model a user with a struct, an associated constructor, and methods.

Beginnerstructs`impl` blocksassociated functions`&self` / `&mut self`

Instructions

Complete the User struct in 11-user-model.rs:

  1. User::new(username: &str, email: &str, age: u8) -> User creates an active user.
  2. display_name(&self) -> String returns "username <email>".
  3. is_adult(&self) -> bool returns true when age >= 18.
  4. deactivate(&mut self) sets active to false.

Expected behavior

alice <alice@example.com>
adult: true
active: false

Optional hints

  • Use field init shorthand: User { username, email, age, active: true }.
  • Convert &str to String with .to_string() or String::from.

Bonus challenge

Use struct update syntax (..user) to create a second user that copies everything except the email.

#[derive(Debug, Clone)]
struct User {
    username: String,
    email: String,
    age: u8,
    active: bool,
}

impl User {
    fn new(username: &str, email: &str, age: u8) -> User {
        User {
            username: username.to_string(),
            email: email.to_string(),
            age,
            active: true,
        }
    }

    fn display_name(&self) -> String {
        format!("{} <{}>", self.username, self.email)
    }

    fn is_adult(&self) -> bool {
        self.age >= 18
    }

    fn deactivate(&mut self) {
        self.active = false;
    }
}

fn main() {
    let mut user = User::new("alice", "alice@example.com", 30);
    println!("{}", user.display_name());
    println!("adult: {}", user.is_adult());
    user.deactivate();
    println!("active: {}", user.active);

    // Bonus: struct update syntax.
    let work = User {
        email: String::from("alice@work.example"),
        ..user.clone()
    };
    println!("{work:?}");
}

#[cfg(test)]
mod tests {
    use super::*;

    #[test]
    fn new_user_is_active() {
        let u = User::new("bob", "bob@example.com", 17);
        assert!(u.active);
        assert!(!u.is_adult());
        assert_eq!(u.display_name(), "bob <bob@example.com>");
    }

    #[test]
    fn deactivate_works() {
        let mut u = User::new("bob", "bob@example.com", 40);
        u.deactivate();
        assert!(!u.active);
    }
}
