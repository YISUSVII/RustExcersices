Exercise 20 · Level 7 — Collections and Strings

Grade Tracker

Store grades per student and compute averages and letter grades.

Intermediate`HashMap<StringVec<u32>>``entry().or_default()``Option`iterator `sum`/`map`sorting

Instructions

Implement GradeTracker (a wrapper around HashMap<String, Vec<u32>>):

  1. GradeTracker::new() -> GradeTracker
  2. add_grade(&mut self, student: &str, grade: u32) -> Result<(), String> — reject grades above 100.
  3. average(&self, student: &str) -> Option<f64> — None for unknown students.
  4. letter(average: f64) -> char — A ≥ 90, B ≥ 80, C ≥ 70, D ≥ 60, otherwise F.
  5. report(&self) -> Vec<(String, f64, char)> — one row per student, sorted by name.

Expected behavior

rejected: grade 120 is out of range (0-100)
ana: 91.0 (A)
ben: 74.5 (C)
cleo: 55.0 (F)
unknown: None

Optional hints

  • self.grades.entry(student.to_string()).or_default().push(grade);
  • Average = sum as f64 / len as f64.
  • match with guards works nicely for letter: a if a >= 90.0 => 'A'.

Bonus challenge

Add best_student(&self) -> Option<&str>.

use std::collections::HashMap;

#[derive(Debug, Default)]
struct GradeTracker {
    grades: HashMap<String, Vec<u32>>,
}

impl GradeTracker {
    fn new() -> GradeTracker {
        GradeTracker::default()
    }

    fn add_grade(&mut self, student: &str, grade: u32) -> Result<(), String> {
        if grade > 100 {
            return Err(format!("grade {grade} is out of range (0-100)"));
        }
        self.grades
            .entry(student.to_string())
            .or_default()
            .push(grade);
        Ok(())
    }

    fn average(&self, student: &str) -> Option<f64> {
        let grades = self.grades.get(student)?;
        if grades.is_empty() {
            return None;
        }
        let sum: u32 = grades.iter().sum();
        Some(sum as f64 / grades.len() as f64)
    }

    fn letter(average: f64) -> char {
        match average {
            a if a >= 90.0 => 'A',
            a if a >= 80.0 => 'B',
            a if a >= 70.0 => 'C',
            a if a >= 60.0 => 'D',
            _ => 'F',
        }
    }

    fn report(&self) -> Vec<(String, f64, char)> {
        let mut rows: Vec<(String, f64, char)> = self
            .grades
            .keys()
            .filter_map(|name| {
                self.average(name)
                    .map(|avg| (name.clone(), avg, GradeTracker::letter(avg)))
            })
            .collect();
        rows.sort_by(|a, b| a.0.cmp(&b.0));
        rows
    }

    fn best_student(&self) -> Option<&str> {
        self.grades
            .keys()
            .filter_map(|name| self.average(name).map(|avg| (name.as_str(), avg)))
            .max_by(|a, b| a.1.total_cmp(&b.1))
            .map(|(name, _)| name)
    }
}

fn main() {
    let mut tracker = GradeTracker::new();
    let entries = [
        ("ana", 95),
        ("ben", 70),
        ("ana", 87),
        ("cleo", 55),
        ("ben", 79),
        ("ben", 120),
    ];
    for (student, grade) in entries {
        if let Err(e) = tracker.add_grade(student, grade) {
            println!("rejected: {e}");
        }
    }

    for (name, avg, letter) in tracker.report() {
        println!("{name}: {avg:.1} ({letter})");
    }
    println!("unknown: {:?}", tracker.average("zed"));
    println!("best: {:?}", tracker.best_student());
}
