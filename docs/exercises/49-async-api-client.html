Async API Client

Build an async API client over a fake in-memory HTTP transport, with timeouts (a select-like future) and retries with backoff.

Ninjaasync traits via boxed futures`Pin<Box<dyn Future + Send + 'a>>`select/timeout futuresretry policieserror enums`async fn` methods

Real projects use a runtime and an HTTP crate. With Tokio + reqwest this is tokio::time::timeout(dur, reqwest::get(url)).await. Here everything is std-only: the block_on/sleep runtime from exercise 47 is provided, and the "network" is an in-memory fake, so you can focus on how timeouts and retries are built from futures.

Instructions

  1. Implement the timeout combinator: 
    struct Timeout<F> { inner: Pin<Box<F>>, deadline: Sleep }
fn timeout<F: Future>(duration: Duration, fut: F) -> Timeout<F>
impl<F: Future> Future for Timeout<F> { type Output = Result<F::Output, Elapsed>; ... }

    Poll inner first (Ready → Ok), then deadline (Ready → Err(Elapsed)).

  2. Use the "async trait" pattern: 
    type BoxFuture<'a, T> = Pin<Box<dyn Future<Output = T> + Send + 'a>>;
trait Transport { fn send(&self, req: Request) -> BoxFuture<'_, Result<Response, TransportError>>; }
  3. Implement Transport for FakeTransport: increment calls, await sleep(latency), fail with TransportError::ConnectionReset while failures_left > 0 (decrementing it), otherwise return the configured route or a 404.
  4. Implement ApiClient<T: Transport> { transport, max_retries, timeout, backoff } with: 
    async fn get(&self, path: &str) -> Result<String, ApiError>
    • map results: Elapsed → ApiError::Timeout, transport error → ApiError::Transport(e), 2xx → Ok(body), otherwise ApiError::Status(code);
    • retry while error.is_retryable() (timeouts, transport errors, 5xx) and attempt < max_retries, sleeping backoff * attempt between attempts.

Expected behavior

  • With 2 initial connection failures and max_retries = 2, get("/users/1") succeeds on the 3rd call.
  • get("/missing") → Err(Status(404)) after exactly 1 call (4xx is not retried).
  • With server latency 100 ms, timeout 20 ms and max_retries = 1 → Err(Timeout) after 2 calls.
  • timeout(50ms, async { 7 }) → Ok(7).

Optional hints

  • Box::pin(async move { ... }) turns an async block into the BoxFuture the trait returns; the '_ lifetime lets the future borrow self.
  • Native async fn in traits is stable since Rust 1.75, but such traits are not object-safe (dyn Transport won't work) — that's why crates like async-trait box the future, as done here.
  • Struct-update syntax ApiClient { timeout: ..., ..ApiClient::new(t) } is handy for tests.

#[tokio::main]
async fn main() -> Result<(), Box<dyn std::error::Error>> {

    for attempt in 0..3 {
        match tokio::time::timeout(Duration::from_secs(2), client.get("https://httpbin.org/get").send()).await {

// ---------- tiny runtime (see exercise 47) ----------

struct ThreadWaker(Thread);

impl Wake for ThreadWaker {
    fn wake(self: Arc<Self>) {
        self.0.unpark();
    }
}

fn block_on<F: Future>(fut: F) -> F::Output {
    let mut fut = pin!(fut);
    let waker = Waker::from(Arc::new(ThreadWaker(thread::current())));
    let mut cx = Context::from_waker(&waker);
    loop {
        if let Poll::Ready(out) = fut.as_mut().poll(&mut cx) {
            return out;
        }
        thread::park();
    }
}

#[derive(Default)]
struct TimerState {
    completed: bool,
    waker: Option<Waker>,
}

struct Sleep {
    state: Arc<Mutex<TimerState>>,
}

fn sleep(duration: Duration) -> Sleep {
    let state = Arc::new(Mutex::new(TimerState::default()));
    let thread_state = Arc::clone(&state);
    thread::spawn(move || {
        thread::sleep(duration);
        let mut s = thread_state.lock().unwrap();
        s.completed = true;
        if let Some(w) = s.waker.take() {
            w.wake();
        }
    });
    Sleep { state }
}

impl Future for Sleep {
    type Output = ();
    fn poll(self: Pin<&mut Self>, cx: &mut Context<'_>) -> Poll<()> {
        let mut s = self.state.lock().unwrap();
        if s.completed {
            Poll::Ready(())
        } else {
            s.waker = Some(cx.waker().clone());
            Poll::Pending
        }
    }
}

#[derive(Debug, PartialEq)]
struct Elapsed;

struct Timeout<F> {
    inner: Pin<Box<F>>,
    deadline: Sleep,
}

fn timeout<F: Future>(duration: Duration, fut: F) -> Timeout<F> {
    Timeout {
        inner: Box::pin(fut),
        deadline: sleep(duration),
    }
}

impl<F: Future> Future for Timeout<F> {
    type Output = Result<F::Output, Elapsed>;

    fn poll(mut self: Pin<&mut Self>, cx: &mut Context<'_>) -> Poll<Self::Output> {
        // Poll the real work first so a just-finished result wins over the deadline.
        if let Poll::Ready(v) = self.inner.as_mut().poll(cx) {
            return Poll::Ready(Ok(v));
        }
        match Pin::new(&mut self.deadline).poll(cx) {
            Poll::Ready(()) => Poll::Ready(Err(Elapsed)),
            Poll::Pending => Poll::Pending,
        }
    }
}

/// An "async trait": methods return boxed futures, which is what the
/// `async-trait` crate generates and what keeps the trait object-safe.
trait Transport {
    fn send(&self, req: Request) -> BoxFuture<'_, Result<Response, TransportError>>;
}

struct FakeTransport {
    routes: HashMap<String, (u16, String)>,
    latency: Duration,
    failures_left: Mutex<usize>,
    calls: AtomicUsize,
}

impl FakeTransport {
    fn new(latency: Duration) -> Self {
        FakeTransport {
            routes: HashMap::new(),
            latency,
            failures_left: Mutex::new(0),
            calls: AtomicUsize::new(0),
        }
    }

    fn route(mut self, path: &str, status: u16, body: &str) -> Self {
        self.routes.insert(path.to_string(), (status, body.to_string()));
        self
    }

    /// The first `n` requests fail with `ConnectionReset`.
    fn failing_first(self, n: usize) -> Self {
        *self.failures_left.lock().unwrap() = n;
        self
    }

    fn calls(&self) -> usize {
        self.calls.load(Ordering::SeqCst)
    }
}

impl Transport for FakeTransport {
    fn send(&self, req: Request) -> BoxFuture<'_, Result<Response, TransportError>> {
        Box::pin(async move {
            self.calls.fetch_add(1, Ordering::SeqCst);
            sleep(self.latency).await;
            let should_fail = {
                let mut left = self.failures_left.lock().unwrap();
                let fail = *left > 0;
                *left = left.saturating_sub(1);
                fail
            };
            if should_fail {
                return Err(TransportError::ConnectionReset);
            }
            let (status, body) = self
                .routes
                .get(&req.path)
                .cloned()
                .unwrap_or((404, format!("{} {} not found", req.method, req.path)));
            Ok(Response { status, body })
        })
    }
}

// ---------- client ----------

#[derive(Debug, PartialEq)]
enum ApiError {
    Timeout,
    Transport(TransportError),
    Status(u16),
}

impl ApiError {
    fn is_retryable(&self) -> bool {
        match self {
            ApiError::Timeout | ApiError::Transport(_) => true,
            ApiError::Status(code) => *code >= 500,
        }
    }
}

struct ApiClient<T: Transport> {
    transport: T,
    max_retries: u32,
    timeout: Duration,
    backoff: Duration,
}

impl<T: Transport> ApiClient<T> {
    fn new(transport: T) -> Self {
        ApiClient {
            transport,
            max_retries: 2,
            timeout: Duration::from_millis(100),
            backoff: Duration::from_millis(5),
        }
    }

    async fn get(&self, path: &str) -> Result<String, ApiError> {
        let mut attempt = 0;
        loop {
            let req = Request {
                method: "GET",
                path: path.to_string(),
            };
            let result = match timeout(self.timeout, self.transport.send(req)).await {
                Err(Elapsed) => Err(ApiError::Timeout),
                Ok(Err(e)) => Err(ApiError::Transport(e)),
                Ok(Ok(resp)) if (200..300).contains(&resp.status) => Ok(resp.body),
                Ok(Ok(resp)) => Err(ApiError::Status(resp.status)),
            };
            match result {
                Err(e) if e.is_retryable() && attempt < self.max_retries => {
                    attempt += 1;
                    println!("  attempt {attempt} failed with {e:?}, retrying...");
                    sleep(self.backoff * attempt).await;
                }
                other => return other,
            }
        }
    }
}
