Exercise 43 · Level 14 — Concurrency

Parallel Counter

Increment a shared counter from many threads safely, first with Arc<Mutex<_>>, then lock-free with AtomicUsize, and sum a slice in parallel with scoped threads.

Intermediate`std::thread::spawn``JoinHandle``Arc<T>``Mutex<T>``AtomicUsize`memory `Ordering``std::thread::scope`

Instructions

  1. Implement fn count_with_mutex(threads: usize, increments: usize) -> usize using Arc<Mutex<usize>> and thread::spawn. Join every handle before reading the result.
  2. Implement fn count_with_atomic(threads: usize, increments: usize) -> usize using AtomicUsize::fetch_add inside thread::scope (no Arc needed).
  3. Implement fn parallel_sum(data: &[u64], threads: usize) -> u64 that splits data into chunks and sums each chunk on its own scoped thread.

Expected behavior

  • count_with_mutex(8, 10_000) → 80000
  • count_with_atomic(8, 10_000) → 80000
  • parallel_sum(&(1..=1_000_000).collect::<Vec<_>>(), 4) → 500000500000
  • parallel_sum(&[], 4) → 0

Optional hints

  • thread::spawn requires 'static closures, so you must Arc::clone and move the clone in.
  • thread::scope(|s| { s.spawn(|| ...); }) lets threads borrow local data because all of them are joined before scope returns.
  • Ordering::Relaxed is enough for a pure counter: you only need atomicity, not ordering with other memory.
  • slice.chunks(n) panics if n == 0 — compute data.len().div_ceil(threads.max(1)).

Bonus challenge

Time the three approaches with std::time::Instant for 8 threads × 1,000,000 increments and explain why the mutex version is slowest.

use std::sync::atomic::{AtomicUsize, Ordering};
use std::sync::{Arc, Mutex};
use std::thread;

/// Each thread increments a shared `Arc<Mutex<usize>>` `increments` times.
fn count_with_mutex(threads: usize, increments: usize) -> usize {
    let counter = Arc::new(Mutex::new(0usize));
    let handles: Vec<_> = (0..threads)
        .map(|_| {
            let counter = Arc::clone(&counter);
            thread::spawn(move || {
                for _ in 0..increments {
                    *counter.lock().unwrap() += 1;
                }
            })
        })
        .collect();

    for h in handles {
        h.join().unwrap();
    }
    let total = *counter.lock().unwrap();
    total
}

/// Same as `count_with_mutex`, but lock-free using an atomic.
fn count_with_atomic(threads: usize, increments: usize) -> usize {
    let counter = AtomicUsize::new(0);
    // Scoped threads may borrow `counter` directly: no Arc needed.
    thread::scope(|s| {
        for _ in 0..threads {
            s.spawn(|| {
                for _ in 0..increments {
                    counter.fetch_add(1, Ordering::Relaxed);
                }
            });
        }
    });
    counter.into_inner()
}

/// Sums `data` by splitting it into `threads` chunks processed in parallel.
fn parallel_sum(data: &[u64], threads: usize) -> u64 {
    if data.is_empty() {
        return 0;
    }
    let chunk_size = data.len().div_ceil(threads.max(1));
    thread::scope(|s| {
        let handles: Vec<_> = data
            .chunks(chunk_size)
            .map(|chunk| s.spawn(move || chunk.iter().sum::<u64>()))
            .collect();
        handles.into_iter().map(|h| h.join().unwrap()).sum()
    })
}

fn main() {
    println!("mutex counter  = {}", count_with_mutex(8, 10_000));
    println!("atomic counter = {}", count_with_atomic(8, 10_000));

    let data: Vec<u64> = (1..=1_000_000).collect();
    println!("parallel sum   = {}", parallel_sum(&data, 4));
}
