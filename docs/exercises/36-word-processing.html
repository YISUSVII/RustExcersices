Exercise 36 · Level 10 — Closures and Iterators

Word Processing

Process text with iterator chains and build your own iterator types and adaptors.

Advanced`split_whitespace``chars``flat_map``windows``peekable`implementing `Iterator`extension traitsreturning `impl Iterator<Item = &str> + '_`

Instructions

  1. normalize_words(text: &str) -> impl Iterator<Item = String> + '_ — split on whitespace, strip leading/trailing non-alphanumeric characters (keep inner '), lowercase, and skip empty results.
  2. word_counts(text: &str) -> BTreeMap<String, usize> — built from normalize_words.
  3. top_n(text: &str, n: usize) -> Vec<(String, usize)> — most frequent words; ties broken alphabetically.
  4. capitalize_words(text: &str) -> String — "hello big world" → "Hello Big World".
  5. long_words(text: &str, min_len: usize) -> impl Iterator<Item = &str> + '_ — words (with surrounding punctuation trimmed) of at least min_len chars; borrows from text (no allocation).
  6. bigrams(text: &str) -> Vec<(String, String)> — consecutive normalized word pairs (hint: windows(2)).
  7. A custom iterator Sentences<'a> that yields trimmed sentences from a &'a str, splitting on ., ! and ? (the terminator stays with the sentence). Provide fn sentences(text: &str) -> Sentences<'_>.
  8. A custom adaptor: struct Chunked<I> { inner: I, size: usize } that groups items into Vec<I::Item> chunks, plus an extension trait 
    trait ChunkedExt: Iterator + Sized {
    fn chunked(self, size: usize) -> Chunked<Self>;
}
impl<I: Iterator> ChunkedExt for I { /* ... */ }

    so you can write words.chunked(3). A size of 0 should be treated as 1.

Expected behavior

  • normalize_words("Hello, world! It's fine.") → ["hello", "world", "it's", "fine"]
  • top_n("a b a c b a", 2) → [("a", 3), ("b", 2)]
  • capitalize_words("hello big world") → "Hello Big World"
  • long_words("a tiny elephant!", 4) → ["tiny", "elephant"]
  • bigrams("one two three") → [("one","two"), ("two","three")]
  • sentences("Hi there. How are you? Great!") → ["Hi there.", "How are you?", "Great!"]
  • (1..=7).chunked(3) → [[1,2,3], [4,5,6], [7]]

Optional hints

  • str::trim_matches(|c: char| !c.is_alphanumeric() && c != '\'').
  • For capitalize_words, take chars(), uppercase the first with to_uppercase(), and chain the rest.
  • In Sentences::next, find the first terminator with find(['.', '!', '?']), slice, and advance the remaining &'a str.
  • In Chunked::next, use self.inner.by_ref().take(self.size).collect() and return None for an empty chunk.

Bonus challenge

Implement a Interleave<A, B> adaptor that alternates items from two iterators until both are exhausted.

use std::collections::BTreeMap;

fn normalize_words(text: &str) -> impl Iterator<Item = String> + '_ {
    text.split_whitespace()
        .map(|w| w.trim_matches(|c: char| !c.is_alphanumeric() && c != '\''))
        .map(|w| w.trim_matches('\''))
        .filter(|w| !w.is_empty())
        .map(str::to_lowercase)
}

fn word_counts(text: &str) -> BTreeMap<String, usize> {
    normalize_words(text).fold(BTreeMap::new(), |mut counts, w| {
        *counts.entry(w).or_insert(0) += 1;
        counts
    })
}

fn top_n(text: &str, n: usize) -> Vec<(String, usize)> {
    let mut counts: Vec<(String, usize)> = word_counts(text).into_iter().collect();
    counts.sort_by(|a, b| b.1.cmp(&a.1).then_with(|| a.0.cmp(&b.0)));
    counts.truncate(n);
    counts
}

fn capitalize_words(text: &str) -> String {
    text.split_whitespace()
        .map(|word| {
            let mut chars = word.chars();
            match chars.next() {
                Some(first) => first.to_uppercase().chain(chars).collect(),
                None => String::new(),
            }
        })
        .collect::<Vec<String>>()
        .join(" ")
}

fn long_words(text: &str, min_len: usize) -> impl Iterator<Item = &str> + '_ {
    text.split_whitespace()
        .map(|w| w.trim_matches(|c: char| !c.is_alphanumeric()))
        .filter(move |w| w.chars().count() >= min_len)
}

fn bigrams(text: &str) -> Vec<(String, String)> {
    let words: Vec<String> = normalize_words(text).collect();
    words
        .windows(2)
        .map(|pair| (pair[0].clone(), pair[1].clone()))
        .collect()
}

struct Sentences<'a> {
    rest: &'a str,
}

fn sentences(text: &str) -> Sentences<'_> {
    Sentences { rest: text }
}

impl<'a> Iterator for Sentences<'a> {
    type Item = &'a str;

    fn next(&mut self) -> Option<Self::Item> {
        let rest = self.rest.trim_start();
        if rest.is_empty() {
            self.rest = rest;
            return None;
        }
        // Terminators are ASCII, so `i + 1` is always a char boundary.
        let end = rest.find(['.', '!', '?']).map_or(rest.len(), |i| i + 1);
        let (sentence, remaining) = rest.split_at(end);
        self.rest = remaining;
        Some(sentence.trim_end())
    }
}

struct Chunked<I> {
    inner: I,
    size: usize,
}

impl<I: Iterator> Iterator for Chunked<I> {
    type Item = Vec<I::Item>;

    fn next(&mut self) -> Option<Self::Item> {
        let chunk: Vec<I::Item> = self.inner.by_ref().take(self.size).collect();
        (!chunk.is_empty()).then_some(chunk)
    }
}

trait ChunkedExt: Iterator + Sized {
    fn chunked(self, size: usize) -> Chunked<Self>;
}

impl<I: Iterator> ChunkedExt for I {
    fn chunked(self, size: usize) -> Chunked<Self> {
        Chunked {
            inner: self,
            size: size.max(1),
        }
    }
}

struct Interleave<A, B> {
    a: A,
    b: B,
    take_a: bool,
}

impl<A, B> Iterator for Interleave<A, B>
where
    A: Iterator,
    B: Iterator<Item = A::Item>,
{
    type Item = A::Item;

    fn next(&mut self) -> Option<Self::Item> {
        let first_try = self.take_a;
        self.take_a = !self.take_a;
        if first_try {
            self.a.next().or_else(|| self.b.next())
        } else {
            self.b.next().or_else(|| self.a.next())
        }
    }
}

fn interleave<A, B>(a: A, b: B) -> Interleave<A::IntoIter, B::IntoIter>
where
    A: IntoIterator,
    B: IntoIterator<Item = A::Item>,
{
    Interleave {
        a: a.into_iter(),
        b: b.into_iter(),
        take_a: true,
    }
}

    let text = "The quick brown fox. The lazy dog sleeps! Does the fox care? No.";

    let words: Vec<String> = normalize_words(text).collect();
    println!("normalized: {words:?}");
    println!("top 3: {:?}", top_n(text, 3));
    println!("capitalized: {}", capitalize_words("hello big world"));
    println!("long words: {:?}", long_words(text, 5).collect::<Vec<_>>());

    for (i, s) in sentences(text).enumerate() {
        println!("sentence {}: {s}", i + 1);
    }

    for chunk in words.iter().chunked(4) {
        println!("chunk: {chunk:?}");
    }

    let mixed: Vec<i32> = interleave(vec![1, 3, 5, 7], vec![2, 4]).collect();
    println!("interleave: {mixed:?}");
}
