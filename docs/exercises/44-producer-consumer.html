Exercise 44 · Level 14 — Concurrency

Producer / Consumer

Pass messages from multiple producer threads to a consumer using std::sync::mpsc channels, both unbounded and bounded.

Intermediate`mpsc::channel``mpsc::sync_channel`cloning `Sender`channel closing semanticsbackpressure`thread::scope`

Instructions

  1. Define struct Message { producer: usize, seq: usize, value: u64 }.
  2. Implement fn run_pipeline(producers: usize, items_per_producer: usize) -> Vec<Message>: - create one channel, clone the Sender for each producer thread; - each producer sends items_per_producer messages with seq = 0..items_per_producer and value = producer * 1000 + seq; - the calling thread consumes until the channel closes and returns all messages.
  3. Implement fn bounded_sum(producers: usize, items_per_producer: u64, bound: usize) -> u64 using mpsc::sync_channel(bound). A dedicated consumer thread sums all received values; each producer sends 1..=items_per_producer.

Expected behavior

  • run_pipeline(5, 20).len() → 100.
  • Messages from the same producer arrive in order (seq 0, 1, 2, …), although producers interleave.
  • bounded_sum(3, 10, 1) → 165.

Optional hints

  • The receiver iterator only ends when every Sender is dropped — including the original one. Forgetting drop(tx) makes the consumer hang forever.
  • sync_channel(bound) blocks send when the buffer is full, which naturally slows fast producers (backpressure).

Bonus challenge

Add a second stage: a "transformer" thread that receives Messages, doubles value, and forwards them to a final consumer through another channel (a 3-stage pipeline).

use std::sync::mpsc;
use std::thread;

#[derive(Debug, Clone, PartialEq)]
struct Message {
    producer: usize,
    seq: usize,
    value: u64,
}

/// Spawns `producers` threads, each sending `items_per_producer` messages
/// over a single channel. The calling thread is the consumer.
fn run_pipeline(producers: usize, items_per_producer: usize) -> Vec<Message> {
    let (tx, rx) = mpsc::channel();

    for id in 0..producers {
        let tx = tx.clone();
        thread::spawn(move || {
            for seq in 0..items_per_producer {
                let msg = Message {
                    producer: id,
                    seq,
                    value: (id * 1000 + seq) as u64,
                };
                if tx.send(msg).is_err() {
                    break; // consumer hung up
                }
            }
        });
    }
    // Drop the original sender so the receiver loop ends once all producers finish.
    drop(tx);

    rx.into_iter().collect()
}

/// Bounded pipeline: producers block when the buffer (capacity `bound`) is full.
/// Returns the sum of all values computed by a dedicated consumer thread.
fn bounded_sum(producers: usize, items_per_producer: u64, bound: usize) -> u64 {
    let (tx, rx) = mpsc::sync_channel::<u64>(bound);
    let consumer = thread::spawn(move || rx.iter().sum::<u64>());

    thread::scope(|s| {
        for _ in 0..producers {
            let tx = tx.clone();
            s.spawn(move || {
                for v in 1..=items_per_producer {
                    tx.send(v).unwrap();
                }
            });
        }
    });
    drop(tx);
    consumer.join().unwrap()
}

fn main() {
    let messages = run_pipeline(3, 4);
    for m in &messages {
        println!("producer {} -> seq {} value {}", m.producer, m.seq, m.value);
    }
    println!("received {} messages", messages.len());

    println!("bounded sum = {}", bounded_sum(4, 100, 2));
}

#[cfg(test)]
mod tests {
    use super::*;

    #[test]
    fn receives_every_message() {
        let msgs = run_pipeline(5, 20);
        assert_eq!(msgs.len(), 100);
    }

    #[test]
    fn per_producer_order_is_preserved() {
