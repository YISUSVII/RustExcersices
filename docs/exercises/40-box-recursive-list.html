Exercise 40 · Level 13 — Smart Pointers

Box Recursive List

Use Box<T> to build recursive data types: a cons list and an arithmetic expression tree.

Intermediate`Box<T>`recursive enumspattern matchingrecursion`Display`

Instructions

  1. Define a cons list: 
    enum List { Cons(i32, Box<List>), Nil }
  2. Implement on List: - fn from_slice(values: &[i32]) -> List - fn len(&self) -> usize - fn sum(&self) -> i32 - impl fmt::Display for List producing (1 (2 (3 Nil))).
  3. Define an expression tree: 
    enum Expr { Num(f64), Add(Box<Expr>, Box<Expr>), Sub(..), Mul(..), Div(..) }
  4. Implement fn eval(&self) -> Result<f64, String> returning Err on division by zero.
  5. Implement impl fmt::Display for Expr printing fully parenthesized expressions, e.g. ((2 + 3) * (10 / 4)).

Expected behavior

  • List::from_slice(&[1, 2, 3]) displays as (1 (2 (3 Nil))), len() == 3, sum() == 6.
  • ((2 + 3) * (10 / 4)) evaluates to Ok(12.5).
  • (1 / (2 - 2)) evaluates to Err("division by zero").

Optional hints

  • Without Box, List would have infinite size; Box gives the recursive field a known size (one pointer).
  • from_slice is a nice fit for iter().rev().fold(Nil, ...).
  • Use the ? operator inside eval to propagate errors from sub-expressions.

Bonus challenge

Add a Neg(Box<Expr>) variant and a simplify(self) -> Expr method that folds constant sub-trees (e.g. Add(Num(1), Num(2)) → Num(3)).

use std::fmt;

#[derive(Debug, PartialEq)]
enum List {
    Cons(i32, Box<List>),
    Nil,
}

use List::{Cons, Nil};

impl List {
    fn from_slice(values: &[i32]) -> List {
        values
            .iter()
            .rev()
            .fold(Nil, |acc, &v| Cons(v, Box::new(acc)))
    }

    fn len(&self) -> usize {
        match self {
            Cons(_, rest) => 1 + rest.len(),
            Nil => 0,
        }
    }

    fn sum(&self) -> i32 {
        match self {
            Cons(v, rest) => v + rest.sum(),
            Nil => 0,
        }
    }
}

impl fmt::Display for List {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        match self {
            Cons(v, rest) => write!(f, "({v} {rest})"),
            Nil => write!(f, "Nil"),
        }
    }
}

#[derive(Debug)]
enum Expr {
    Num(f64),
    Add(Box<Expr>, Box<Expr>),
    Sub(Box<Expr>, Box<Expr>),
    Mul(Box<Expr>, Box<Expr>),
    Div(Box<Expr>, Box<Expr>),
}

impl Expr {
    fn eval(&self) -> Result<f64, String> {
        match self {
            Expr::Num(n) => Ok(*n),
            Expr::Add(a, b) => Ok(a.eval()? + b.eval()?),
            Expr::Sub(a, b) => Ok(a.eval()? - b.eval()?),
            Expr::Mul(a, b) => Ok(a.eval()? * b.eval()?),
            Expr::Div(a, b) => {
                let divisor = b.eval()?;
                if divisor == 0.0 {
                    Err("division by zero".to_string())
                } else {
                    Ok(a.eval()? / divisor)
                }
            }
        }
    }
}

impl fmt::Display for Expr {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        match self {
            Expr::Num(n) => write!(f, "{n}"),
            Expr::Add(a, b) => write!(f, "({a} + {b})"),
            Expr::Sub(a, b) => write!(f, "({a} - {b})"),
            Expr::Mul(a, b) => write!(f, "({a} * {b})"),
            Expr::Div(a, b) => write!(f, "({a} / {b})"),
        }
    }
}

fn num(n: f64) -> Box<Expr> {
    Box::new(Expr::Num(n))
}

fn main() {
    let list = List::from_slice(&[1, 2, 3]);
    println!("list = {list}");
    println!("len = {}, sum = {}", list.len(), list.sum());

    // (2 + 3) * (10 / 4)
    let expr = Expr::Mul(
        Box::new(Expr::Add(num(2.0), num(3.0))),
        Box::new(Expr::Div(num(10.0), num(4.0))),
    );
    println!("{expr} = {:?}", expr.eval());

    let bad = Expr::Div(num(1.0), Box::new(Expr::Sub(num(2.0), num(2.0))));
    println!("{bad} = {:?}", bad.eval());
}
