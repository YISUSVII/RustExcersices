Exercise 51 · Level 16 — Macros

Repetition Macros

Use macro repetitions to generate whole items: an enum with Display/FromStr, a config struct with defaults and getters, and a family of test functions.

Advanced`ident``ty``literal``path``expr` fragmentsitem-generating macrosnested repetitions`stringify!`code generation for tests

Instructions

  1. string_enum! { Name { Variant => "text", ... } } must generate: - enum Name { Variant, ... } deriving Debug, Clone, Copy, PartialEq, Eq - const ALL: &'static [Name] listing every variant - impl fmt::Display printing each variant's text - impl FromStr (Err = String) parsing the text back (unknown → Err)
  2. config_struct! { Name { field: Type = default_expr, ... } } must generate: - the struct, impl Default using the default expressions - one getter per field: fn field(&self) -> &Type - fn field_names() -> &'static [&'static str]
  3. test_cases! { func; case_name: (input, expected), ... } must generate one #[test] fn case_name() per case asserting func(input) == expected.
  4. Use them for Color { Red, Green, Blue }, ServerConfig { host, port, workers }, and parse_color tests.

Expected behavior

  • Color::ALL.len() → 3; Color::Green.to_string() → "green"; "pink".parse::<Color>() → Err("unknown Color: \"pink\"").
  • ServerConfig::default().port() → &8080; ServerConfig::field_names() → ["host", "port", "workers"].
  • cargo test/rustc --test lists parses_red, parses_blue, rejects_unknown, is_case_sensitive as separate tests.

Optional hints

  • A macro can expand to several items at once — just write them one after another in the transcriber.
  • Getter names can reuse the field identifiers: $( fn $field(&self) -> &$ty { &self.$field } )+.
  • stringify!($field) turns an identifier into a &'static str literal.
  • In FromStr, literal patterns work in match: $($text => Ok($name::$variant),)+.

Bonus challenge

Add per-field validation to config_struct! (e.g. port: u16 = 8080 => |p| *p > 0) generating a fn validate(&self) -> Result<(), String>, and add a with_<field> builder method per field (hint: identifier concatenation needs the paste crate — or a different design!).

use std::fmt;
use std::str::FromStr;

/// Generates a fieldless enum with `Display`, `FromStr`, and an `ALL` constant.
///
/// ```ignore
/// string_enum! { Color { Red => "red", Green => "green" } }
/// ```
macro_rules! string_enum {
    ($name:ident { $($variant:ident => $text:literal),+ $(,)? }) => {
        #[derive(Debug, Clone, Copy, PartialEq, Eq)]
        enum $name {
            $($variant),+
        }

        impl $name {
            const ALL: &'static [$name] = &[$($name::$variant),+];
        }

        impl fmt::Display for $name {
            fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
                let s = match self {
                    $($name::$variant => $text),+
                };
                f.write_str(s)
            }
        }

        impl FromStr for $name {
            type Err = String;
            fn from_str(s: &str) -> Result<Self, Self::Err> {
                match s {
                    $($text => Ok($name::$variant),)+
                    other => Err(format!("unknown {}: {other:?}", stringify!($name))),
                }
            }
        }
    };
}

/// Generates a struct with a `Default` impl (from the given defaults),
/// a getter per field, and a `field_names()` list.
macro_rules! config_struct {
    ($name:ident { $($field:ident : $ty:ty = $default:expr),+ $(,)? }) => {
        #[derive(Debug, Clone, PartialEq)]
        struct $name {
            $($field: $ty),+
        }

        impl Default for $name {
            fn default() -> Self {
                $name { $($field: $default),+ }
            }
        }

        impl $name {
            $(
                fn $field(&self) -> &$ty {
                    &self.$field
                }
            )+

            fn field_names() -> &'static [&'static str] {
                &[$(stringify!($field)),+]
            }
        }
    };
}

/// Generates one `#[test]` function per case: `name: (input, expected)`.
#[allow(unused_macros)]
macro_rules! test_cases {
    ($func:path; $($name:ident : ($input:expr, $expected:expr)),+ $(,)?) => {
        $(
            #[test]
            fn $name() {
                assert_eq!($func($input), $expected, "case {}", stringify!($name));
            }
        )+
    };
}

    Color {
        Red => "red",
        Green => "green",
        Blue => "blue",
    }
}
