Exercise 38 · Level 12 — Testing

Unit Tests

Write a thorough unit-test suite for already-implemented functions, covering happy paths, edge cases, error values and expected panics.

Intermediate`#[cfg(test)]``#[test]``assert!``assert_eq!``assert_ne!`custom failure messages`#[should_panic(expected = "...")]`tests returning `Result<()E>``use super::*`

Instructions

The starter file already contains working implementations of:

  • is_palindrome(s: &str) -> bool — ignores case and non-alphanumeric characters.
  • divide(a: i32, b: i32) -> Result<i32, String> — errors on division by zero and on overflow (i32::MIN / -1).
  • parse_rgb(hex: &str) -> Result<(u8, u8, u8), String> — parses "#rrggbb" (the # is optional).
  • BoundedStack<T> with new(capacity: usize) -> Self (panics if capacity == 0), push(&mut self, item: T) -> Result<(), T>, pop(&mut self) -> Option<T>, peek(&self) -> Option<&T>, len(&self) -> usize, is_empty(&self) -> bool.

Your job is to replace every todo!() in the #[cfg(test)] mod tests block:

  1. palindrome_ignores_case_and_punctuation — at least one positive and one negative case, with a custom message on the negative assertion.
  2. palindrome_edge_cases — empty string, punctuation-only string, a unicode word.
  3. divide_happy_path_truncates_toward_zero — check 7 / 2 and -7 / 2.
  4. divide_reports_errors — division by zero (exact message) and overflow (message contains "overflow").
  5. parse_rgb_with_and_without_hash — make the test return Result<(), String> and use ? instead of unwrap().
  6. parse_rgb_rejects_bad_input — wrong length, non-hex digits, non-ASCII input.
  7. stack_is_lifo_and_bounded — fill the stack, assert push hands the item back when full, then check peek, len and LIFO pop order down to None.
  8. stack_with_zero_capacity_panics — add #[should_panic(expected = "capacity must be greater than zero")].

Run your tests with:

rustc --edition 2021 --test 38-unit-tests.rs -o unit_tests && ./unit_tests
# or, inside a Cargo project:
cargo test

Expected behavior

  • Before you start: test result: FAILED. 0 passed; 8 failed.
  • When you finish: test result: ok. 8 passed; 0 failed.
  • Temporarily break an implementation (e.g. change checked_div to /) and confirm at least one test fails with a clear message.

Optional hints

  • assert!(cond, "message with {}", value) — the message is only built when the assertion fails.
  • assert_eq!(left, right) prints both values on failure; your types need Debug + PartialEq.
  • unwrap_err() is handy to inspect an error message: assert!(err.contains("overflow")).
  • A test returning Result fails if it returns Err; you cannot combine that with #[should_panic].
  • Filter tests by name: ./unit_tests divide runs only tests whose name contains divide.

Bonus challenge

  • Add a deliberately slow test marked #[ignore] and run it with ./unit_tests --ignored.
  • Move the tests into a Cargo project's tests/ folder as an integration test — what has to become pub?

/// Returns true if `s` reads the same forwards and backwards,
/// ignoring case and any non-alphanumeric characters.
fn is_palindrome(s: &str) -> bool {
    let cleaned: Vec<char> = s
        .chars()
        .filter(|c| c.is_alphanumeric())
        .flat_map(char::to_lowercase)
        .collect();
    cleaned.iter().eq(cleaned.iter().rev())
}

/// Integer division that reports errors instead of panicking.
fn divide(a: i32, b: i32) -> Result<i32, String> {
    if b == 0 {
        return Err("division by zero".to_string());
    }
    a.checked_div(b)
        .ok_or_else(|| format!("overflow dividing {a} by {b}"))
}

/// Parses a hex color like "#ff8800" (the `#` is optional).
fn parse_rgb(hex: &str) -> Result<(u8, u8, u8), String> {
    let digits = hex.strip_prefix('#').unwrap_or(hex);
    if digits.len() != 6 || !digits.is_ascii() {
        return Err(format!("expected 6 hex digits, got '{hex}'"));
    }
    let channel = |range: std::ops::Range<usize>| {
        u8::from_str_radix(&digits[range], 16).map_err(|e| format!("invalid hex in '{hex}': {e}"))
    };
    Ok((channel(0..2)?, channel(2..4)?, channel(4..6)?))
}

/// A stack with a fixed maximum size.
#[derive(Debug)]
struct BoundedStack<T> {
    items: Vec<T>,
    capacity: usize,
}

impl<T> BoundedStack<T> {
    /// # Panics
    /// Panics if `capacity` is zero — that is a programming error, not a runtime condition.
    fn new(capacity: usize) -> Self {
        assert!(capacity > 0, "capacity must be greater than zero");
        Self {
            items: Vec::with_capacity(capacity),
            capacity,
        }
    }

    /// Pushes an item, handing it back as `Err(item)` when the stack is full.
    fn push(&mut self, item: T) -> Result<(), T> {
        if self.items.len() == self.capacity {
            return Err(item);
        }
        self.items.push(item);
        Ok(())
    }

    fn pop(&mut self) -> Option<T> {
        self.items.pop()
    }

    fn peek(&self) -> Option<&T> {
        self.items.last()
    }

    fn len(&self) -> usize {
        self.items.len()
    }

    fn is_empty(&self) -> bool {
        self.items.is_empty()
    }
}

fn main() {
    println!(
        "is_palindrome(\"A man, a plan, a canal: Panama\") = {}",
        is_palindrome("A man, a plan, a canal: Panama")
    );
    println!("divide(7, 2) = {:?}", divide(7, 2));
    println!("divide(1, 0) = {:?}", divide(1, 0));
    println!("divide(i32::MIN, -1) = {:?}", divide(i32::MIN, -1));
    println!("parse_rgb(\"#ff8800\") = {:?}", parse_rgb("#ff8800"));
    println!("parse_rgb(\"#zz0000\") = {:?}", parse_rgb("#zz0000"));

    let mut stack = BoundedStack::new(2);
    println!("push 1 -> {:?}", stack.push(1));
    println!("push 2 -> {:?}", stack.push(2));
    println!("push 3 -> {:?} (full)", stack.push(3));
    println!("peek -> {:?}, len -> {}", stack.peek(), stack.len());
    println!(
        "pop -> {:?}, pop -> {:?}, pop -> {:?}",
        stack.pop(),
        stack.pop(),
        stack.pop()
    );
    println!("is_empty -> {}", stack.is_empty());
}

#[cfg(test)]
mod tests {
    use super::*;

    // --- is_palindrome -------------------------------------------------------

    #[test]
    fn palindrome_ignores_case_and_punctuation() {
        assert!(is_palindrome("A man, a plan, a canal: Panama"));
        assert!(is_palindrome("Racecar"));
        assert!(!is_palindrome("hello"), "'hello' is not a palindrome");
    }

    #[test]
    fn palindrome_edge_cases() {
        assert!(is_palindrome(""), "empty string counts as a palindrome");
        assert!(is_palindrome("!!!"), "punctuation-only reduces to empty");
        assert!(is_palindrome("ÉtÉ"), "unicode letters are supported");
    }

    // --- divide --------------------------------------------------------------

    #[test]
    fn divide_happy_path_truncates_toward_zero() {
        assert_eq!(divide(7, 2), Ok(3));
        assert_eq!(divide(-7, 2), Ok(-3));
    }

    #[test]
    fn divide_reports_errors() {
        assert_eq!(divide(1, 0), Err("division by zero".to_string()));
        let err = divide(i32::MIN, -1).unwrap_err();
        assert!(err.contains("overflow"), "unexpected message: {err}");
    }

    // --- parse_rgb: Result-returning tests can use `?` -----------------------

    #[test]
    fn parse_rgb_with_and_without_hash() -> Result<(), String> {
        assert_eq!(parse_rgb("#ff8800")?, (255, 136, 0));
        assert_eq!(parse_rgb("000000")?, (0, 0, 0));
        Ok(())
    }

    #[test]
    fn parse_rgb_rejects_bad_input() {
        assert!(parse_rgb("#fff").is_err(), "too short");
        assert!(parse_rgb("#zz0000").is_err(), "not hex");
        assert!(parse_rgb("#ffé000").is_err(), "non-ascii must not panic");
    }

    // --- BoundedStack --------------------------------------------------------

    #[test]
    fn stack_is_lifo_and_bounded() -> Result<(), i32> {
        let mut stack = BoundedStack::new(2);
        assert!(stack.is_empty());
        stack.push(1)?;
        stack.push(2)?;
        assert_eq!(stack.push(3), Err(3), "full stack must hand the item back");
        assert_eq!(stack.peek(), Some(&2));
        assert_eq!(stack.len(), 2);
        assert_eq!(stack.pop(), Some(2));
        assert_eq!(stack.pop(), Some(1));
        assert_eq!(stack.pop(), None);
        Ok(())
    }

    #[test]
    #[should_panic(expected = "capacity must be greater than zero")]
    fn stack_with_zero_capacity_panics() {
        let _ = BoundedStack::<u8>::new(0);
    }
}
