Exercise 31 · Level 9 — Generics, Traits and Lifetimes

Repository Abstraction

Design a generic Repository<T> trait and an in-memory implementation that works for any entity type with an id.

Advancedgeneric traitsassociated types vs generic parameterstrait bounds`HashMap``Result` with a custom error enumborrowing in return types

Instructions

  1. Define an Entity trait for things that can be stored: 
    trait Entity: Clone {
    fn id(&self) -> u32;
}
  2. Define a custom error type: 
    #[derive(Debug, PartialEq)]
enum RepoError { NotFound(u32), AlreadyExists(u32) }

    and implement std::fmt::Display for it.

  3. Define the generic trait: 
    trait Repository<T: Entity> {
    fn insert(&mut self, item: T) -> Result<(), RepoError>;
    fn get(&self, id: u32) -> Option<&T>;
    fn update(&mut self, item: T) -> Result<T, RepoError>; // returns the old value
    fn delete(&mut self, id: u32) -> Result<T, RepoError>;
    fn all(&self) -> Vec<&T>;                              // sorted by id
    fn count(&self) -> usize { self.all().len() }          // default method
}
  4. Implement InMemoryRepository<T> backed by HashMap<u32, T> with InMemoryRepository::new(), and implement Repository<T> for it for every T: Entity.
  5. Create two entity types, User { id, name } and Product { id, title, price_cents }, and use the same repository code for both.
  6. Write a generic helper find_where<T, R, P>(repo: &R, predicate: P) -> Vec<&T> with a where clause (T: Entity, R: Repository<T>, P: Fn(&T) -> bool).

Expected behavior

  • Inserting a user twice with id 1 => Err(RepoError::AlreadyExists(1))
  • get(99) => None
  • delete(99) => Err(RepoError::NotFound(99))
  • update(user) returns the previous version of that user
  • all() returns items ordered by id
  • find_where(&products, |p| p.price_cents < 1000) returns only cheap products

Optional hints

  • HashMap::entry(id) with Entry::Occupied / Entry::Vacant makes insert clean.
  • HashMap::insert returns the old value — handy for update once you've checked the key exists.
  • ok_or(RepoError::NotFound(id)) turns an Option into a Result.

Bonus challenge

Make the id type generic too: trait Entity { type Id: Eq + Hash + Ord + Copy; fn id(&self) -> Self::Id; } and adapt the repository.

use std::collections::hash_map::Entry;
use std::collections::HashMap;
use std::fmt;

trait Entity: Clone {
    fn id(&self) -> u32;
}

#[derive(Debug, PartialEq)]
enum RepoError {
    NotFound(u32),
    AlreadyExists(u32),
}

impl fmt::Display for RepoError {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        match self {
            RepoError::NotFound(id) => write!(f, "entity with id {id} was not found"),
            RepoError::AlreadyExists(id) => write!(f, "entity with id {id} already exists"),
        }
    }
}

impl std::error::Error for RepoError {}

trait Repository<T: Entity> {
    fn insert(&mut self, item: T) -> Result<(), RepoError>;
    fn get(&self, id: u32) -> Option<&T>;
    fn update(&mut self, item: T) -> Result<T, RepoError>;
    fn delete(&mut self, id: u32) -> Result<T, RepoError>;
    fn all(&self) -> Vec<&T>;

    fn count(&self) -> usize {
        self.all().len()
    }
}

struct InMemoryRepository<T> {
    items: HashMap<u32, T>,
}

impl<T> InMemoryRepository<T> {
    fn new() -> Self {
        Self {
            items: HashMap::new(),
        }
    }
}

impl<T: Entity> Repository<T> for InMemoryRepository<T> {
    fn insert(&mut self, item: T) -> Result<(), RepoError> {
        match self.items.entry(item.id()) {
            Entry::Occupied(e) => Err(RepoError::AlreadyExists(*e.key())),
            Entry::Vacant(e) => {
                e.insert(item);
                Ok(())
            }
        }
    }

    fn get(&self, id: u32) -> Option<&T> {
        self.items.get(&id)
    }

    fn update(&mut self, item: T) -> Result<T, RepoError> {
        let slot = self
            .items
            .get_mut(&item.id())
            .ok_or(RepoError::NotFound(item.id()))?;
        Ok(std::mem::replace(slot, item))
    }

    fn delete(&mut self, id: u32) -> Result<T, RepoError> {
        self.items.remove(&id).ok_or(RepoError::NotFound(id))
    }

    fn all(&self) -> Vec<&T> {
        let mut items: Vec<&T> = self.items.values().collect();
        items.sort_by_key(|item| item.id());
        items
    }

    // Override the default: no need to build a Vec just to count.
    fn count(&self) -> usize {
        self.items.len()
    }
}

#[derive(Debug, Clone, PartialEq)]
struct User {
    id: u32,
    name: String,
}

#[derive(Debug, Clone, PartialEq)]
struct Product {
    id: u32,
    title: String,
    price_cents: u64,
}

impl Entity for User {
    fn id(&self) -> u32 {
        self.id
    }
}

impl Entity for Product {
    fn id(&self) -> u32 {
        self.id
    }
}

fn find_where<'r, T, R, P>(repo: &'r R, predicate: P) -> Vec<&'r T>
where
    T: Entity,
    R: Repository<T>,
    P: Fn(&T) -> bool,
{
    repo.all()
        .into_iter()
        .filter(|item| predicate(item))
        .collect()
}

fn user(id: u32, name: &str) -> User {
    User {
        id,
        name: name.to_string(),
    }
}

fn product(id: u32, title: &str, price_cents: u64) -> Product {
    Product {
        id,
        title: title.to_string(),
        price_cents,
    }
}

fn run() -> Result<(), RepoError> {
    let mut users = InMemoryRepository::new();
    users.insert(user(2, "Bob"))?;
    users.insert(user(1, "Ada"))?;

    if let Err(e) = users.insert(user(1, "Duplicate")) {
        println!("Expected error: {e}");
    }

    let old = users.update(user(2, "Robert"))?;
    println!(
        "Renamed {} -> {:?}",
        old.name,
        users.get(2).map(|u| &u.name)
    );
    println!("All users: {:?}", users.all());
    println!("get(99): {:?}", users.get(99));
    println!("delete(99): {:?}", users.delete(99));

    let mut products = InMemoryRepository::new();
    products.insert(product(10, "Keyboard", 4_999))?;
    products.insert(product(11, "Sticker", 199))?;
    products.insert(product(12, "Mug", 899))?;

    let cheap = find_where(&products, |p: &Product| p.price_cents < 1_000);
    let titles: Vec<&str> = cheap.iter().map(|p| p.title.as_str()).collect();
    println!("Cheap products: {titles:?}");

    let removed = products.delete(10)?;
    println!(
        "Removed {}; {} products left",
        removed.title,
        products.count()
    );
    Ok(())
}

fn main() {
    if let Err(e) = run() {
        eprintln!("error: {e}");
    }
}
