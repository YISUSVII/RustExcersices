Exercise 25 · Level 8 — Error Handling

Parsing with `Result`

Turn untrusted strings into typed values, propagating or converting errors instead of panicking.

Intermediate`str::parse``ParseIntError`collecting into `Result<Vec<_>_>``?``map_err``ok()?` on `Option`

Instructions

  1. Implement parse_numbers(input: &str) -> Result<Vec<i64>, ParseIntError> for comma-separated values like "1, 2, 3" (trim each item). The first bad item fails the whole parse.
  2. Implement sum_numbers(input: &str) -> Result<i64, ParseIntError> using parse_numbers and ?.
  3. Implement parse_duration(input: &str) -> Result<u64, String> that converts strings like "1h30m15s", "45s" or "2m" into seconds: - units are h, m, s; - errors (as String) for empty input, a missing number before a unit, an unknown unit, or trailing digits without a unit.
  4. Implement parse_hex_color(input: &str) -> Option<(u8, u8, u8)> for "#RRGGBB" using u8::from_str_radix(..).ok()?.

Expected behavior

parse_numbers("1, 2, 3") = Ok([1, 2, 3])
sum_numbers("10,20,x") = Err(ParseIntError { kind: InvalidDigit })
parse_duration("1h30m15s") = Ok(5415)
parse_duration("10x") = Err("unknown unit 'x'")
parse_duration("15") = Err("missing unit after 15")
parse_hex_color("#FF8000") = Some((255, 128, 0))
parse_hex_color("FF8000") = None

Optional hints

  • input.split(',').map(|s| s.trim().parse::<i64>()).collect::<Result<Vec<_>, _>>()
  • For durations, accumulate digits in a String and flush them when you meet a unit letter.
  • str::strip_prefix('#') returns an Option.

Bonus challenge

Make parse_duration reject repeated units such as "1h2h".

use std::num::ParseIntError;

fn parse_numbers(input: &str) -> Result<Vec<i64>, ParseIntError> {
    input.split(',').map(|s| s.trim().parse::<i64>()).collect()
}

fn sum_numbers(input: &str) -> Result<i64, ParseIntError> {
    let numbers = parse_numbers(input)?;
    Ok(numbers.iter().sum())
}

fn parse_duration(input: &str) -> Result<u64, String> {
    let input = input.trim();
    if input.is_empty() {
        return Err("empty duration".to_string());
    }

    let mut total = 0;
    let mut digits = String::new();
    let mut seen_units = String::new();

    for c in input.chars() {
        if c.is_ascii_digit() {
            digits.push(c);
            continue;
        }
        let multiplier = match c {
            'h' => 3600,
            'm' => 60,
            's' => 1,
            other => return Err(format!("unknown unit '{other}'")),
        };
        if seen_units.contains(c) {
            return Err(format!("repeated unit '{c}'"));
        }
        seen_units.push(c);
        let value: u64 = digits
            .parse()
            .map_err(|_| format!("missing number before '{c}'"))?;
        total = value
            .checked_mul(multiplier)
            .and_then(|secs| secs.checked_add(total))
            .ok_or_else(|| "duration too large".to_string())?;
        digits.clear();
    }

    if !digits.is_empty() {
        return Err(format!("missing unit after {digits}"));
    }
    Ok(total)
}

fn parse_hex_color(input: &str) -> Option<(u8, u8, u8)> {
    let hex = input.strip_prefix('#')?;
    if hex.len() != 6 || !hex.is_ascii() {
        return None;
    }
    let r = u8::from_str_radix(&hex[0..2], 16).ok()?;
    let g = u8::from_str_radix(&hex[2..4], 16).ok()?;
    let b = u8::from_str_radix(&hex[4..6], 16).ok()?;
    Some((r, g, b))
}

fn main() {
    println!(
        "parse_numbers(\"1, 2, 3\") = {:?}",
        parse_numbers("1, 2, 3")
    );
    println!("sum_numbers(\"10,20,x\") = {:?}", sum_numbers("10,20,x"));
    for d in ["1h30m15s", "10x", "15"] {
        println!("parse_duration({d:?}) = {:?}", parse_duration(d));
    }
    for c in ["#FF8000", "FF8000"] {
        println!("parse_hex_color({c:?}) = {:?}", parse_hex_color(c));
    }
}
