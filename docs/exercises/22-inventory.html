Exercise 22 · Level 7 — Collections and Strings

Inventory Manager

Manage stock levels in a HashMap with safe updates and reports.

Intermediate`HashMap<StringItem>``entry` API`get_mut``Result` for invalid updatesiterator aggregation

Instructions

Items are Item { quantity: u32, price_cents: u64 }. Implement Inventory:

  1. Inventory::new() -> Inventory
  2. add(&mut self, name: &str, quantity: u32, price_cents: u64) — insert a new item or increase the quantity of an existing one (and update its price).
  3. remove(&mut self, name: &str, quantity: u32) -> Result<u32, String> — returns the remaining quantity; errors for unknown items or insufficient stock. Remove the entry when it reaches 0.
  4. total_value(&self) -> u64 — sum of quantity * price_cents.
  5. low_stock(&self, threshold: u32) -> Vec<&str> — names with quantity < threshold, sorted alphabetically.

Expected behavior

remove 3 apples -> Ok(7)
remove 5 kiwis -> Err("unknown item: kiwi")
remove 9 pears -> Err("not enough pear: have 4, requested 9")
remove 4 pears -> Ok(0)
total value: $9.80
low stock (< 8): ["apple", "banana"]

Optional hints

  • let item = self.items.get_mut(name).ok_or_else(|| format!(...))?;
  • low_stock can return &str borrowed from the map keys — no clones needed.

Bonus challenge

Format total_value as dollars with a format_cents(cents: u64) -> String helper.

use std::collections::HashMap;

#[derive(Debug, Clone, PartialEq)]
struct Item {
    quantity: u32,
    price_cents: u64,
}

#[derive(Debug, Default)]
struct Inventory {
    items: HashMap<String, Item>,
}

impl Inventory {
    fn new() -> Inventory {
        Inventory::default()
    }

    fn add(&mut self, name: &str, quantity: u32, price_cents: u64) {
        let item = self.items.entry(name.to_string()).or_insert(Item {
            quantity: 0,
            price_cents,
        });
        item.quantity += quantity;
        item.price_cents = price_cents;
    }

    fn remove(&mut self, name: &str, quantity: u32) -> Result<u32, String> {
        let item = self
            .items
            .get_mut(name)
            .ok_or_else(|| format!("unknown item: {name}"))?;
        if quantity > item.quantity {
            return Err(format!(
                "not enough {name}: have {}, requested {quantity}",
                item.quantity
            ));
        }
        item.quantity -= quantity;
        let remaining = item.quantity;
        if remaining == 0 {
            self.items.remove(name);
        }
        Ok(remaining)
    }

    fn total_value(&self) -> u64 {
        self.items
            .values()
            .map(|item| u64::from(item.quantity) * item.price_cents)
            .sum()
    }

    fn low_stock(&self, threshold: u32) -> Vec<&str> {
        let mut names: Vec<&str> = self
            .items
            .iter()
            .filter(|(_, item)| item.quantity < threshold)
            .map(|(name, _)| name.as_str())
            .collect();
        names.sort_unstable();
        names
    }
}

fn format_cents(cents: u64) -> String {
    format!("${}.{:02}", cents / 100, cents % 100)
}

fn main() {
    let mut inv = Inventory::new();
    inv.add("apple", 10, 50);
    inv.add("banana", 6, 25);
    inv.add("pear", 4, 80);
    inv.add("orange", 12, 40);

    println!("remove 3 apples -> {:?}", inv.remove("apple", 3));
    println!("remove 5 kiwis -> {:?}", inv.remove("kiwi", 5));
    println!("remove 9 pears -> {:?}", inv.remove("pear", 9));
    println!("remove 4 pears -> {:?}", inv.remove("pear", 4));
    println!("total value: {}", format_cents(inv.total_value()));
    println!("low stock (< 8): {:?}", inv.low_stock(8));
}
