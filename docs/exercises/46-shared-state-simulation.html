Exercise 46 · Level 14 — Concurrency

Shared State Simulation

Simulate concurrent bank transfers without data races or deadlocks, and recover from a poisoned mutex.

Ninja`Mutex<T>``RwLock<T>`lock orderingdeadlock avoidance`PoisonError::into_inner``thread::scope``Arc<T>`

Instructions

  1. Define: 
    enum TransferError { SameAccount, UnknownAccount(usize), InsufficientFunds { account: usize, balance: i64 } }
struct Bank { accounts: Vec<Mutex<i64>>, log: RwLock<Vec<String>> }
  2. Implement fn lock_recover<T>(m: &Mutex<T>) -> MutexGuard<'_, T> that returns the guard even if the mutex is poisoned.
  3. Implement on Bank: - fn new(balances: &[i64]) -> Bank - fn transfer(&self, from: usize, to: usize, amount: i64) -> Result<(), TransferError> — always lock the lower index first, then append "{from} -> {to}: {amount}" to log (write lock). - fn balance(&self, idx: usize) -> i64 - fn total(&self) -> i64 — lock all accounts (in order) for a consistent snapshot. - fn log_len(&self) -> usize (read lock).
  4. Implement fn simulate(bank: &Bank, threads: usize, rounds: usize) that uses thread::scope to run many transfers in both directions across accounts, plus an "auditor" thread reading the log.
  5. In main, demonstrate poisoning: a thread panics while holding a lock; show is_poisoned() and recover the value with lock_recover.

Expected behavior

  • After simulate(&bank, 8, 1_000) on 4 accounts of 1000, bank.total() is still 4000 and the program never hangs.
  • transfer(0, 0, 1) → Err(SameAccount); overdrafts → Err(InsufficientFunds { .. }).
  • A poisoned account mutex is still usable through lock_recover.

Optional hints

  • Deadlock: thread 1 locks A then waits for B while thread 2 locks B then waits for A. A global lock order (by index) makes that impossible.
  • m.lock().unwrap_or_else(PoisonError::into_inner) recovers the guard from a poisoned lock.
  • RwLock allows many simultaneous readers or one writer — good for an audit log read far more often than written.

Bonus challenge

Replace per-account Mutex<i64> with AtomicI64 and implement transfer with compare_exchange loops. Which invariants become harder to guarantee (hint: total() snapshots)?

use std::sync::{Arc, Mutex, MutexGuard, PoisonError, RwLock};
use std::thread;

#[derive(Debug, PartialEq)]
enum TransferError {
    SameAccount,
    UnknownAccount(usize),
    InsufficientFunds { account: usize, balance: i64 },
}

struct Bank {
    accounts: Vec<Mutex<i64>>,
    /// Many readers (auditors), occasional writer (transfers).
    log: RwLock<Vec<String>>,
}

/// Locks a mutex, recovering the data even if another thread panicked while holding it.
fn lock_recover<T>(m: &Mutex<T>) -> MutexGuard<'_, T> {
    m.lock().unwrap_or_else(PoisonError::into_inner)
}

impl Bank {
    fn new(balances: &[i64]) -> Bank {
        Bank {
            accounts: balances.iter().map(|&b| Mutex::new(b)).collect(),
            log: RwLock::new(Vec::new()),
        }
    }

    /// Moves `amount` from `from` to `to`. Always locks the lower index first so two
    /// opposite transfers (A→B and B→A) can never deadlock.
    fn transfer(&self, from: usize, to: usize, amount: i64) -> Result<(), TransferError> {
        if from == to {
            return Err(TransferError::SameAccount);
        }
        for idx in [from, to] {
            if idx >= self.accounts.len() {
                return Err(TransferError::UnknownAccount(idx));
            }
        }

        let (first, second) = if from < to { (from, to) } else { (to, from) };
        let mut g1 = lock_recover(&self.accounts[first]);
        let mut g2 = lock_recover(&self.accounts[second]);
        let (src, dst) = if from < to {
            (&mut *g1, &mut *g2)
        } else {
            (&mut *g2, &mut *g1)
        };

        if *src < amount {
            return Err(TransferError::InsufficientFunds {
                account: from,
                balance: *src,
            });
        }
        *src -= amount;
        *dst += amount;
        drop((g1, g2));

        self.log
            .write()
            .unwrap_or_else(PoisonError::into_inner)
            .push(format!("{from} -> {to}: {amount}"));
        Ok(())
    }

    fn balance(&self, idx: usize) -> i64 {
        *lock_recover(&self.accounts[idx])
    }

    /// Total money in the bank. Locks all accounts in index order for a consistent snapshot.
    fn total(&self) -> i64 {
        let guards: Vec<_> = self.accounts.iter().map(lock_recover).collect();
        guards.iter().map(|g| **g).sum()
    }

    fn log_len(&self) -> usize {
        self.log.read().unwrap_or_else(PoisonError::into_inner).len()
    }
}

/// Runs `rounds` transfers per thread across all account pairs in both directions.
fn simulate(bank: &Bank, threads: usize, rounds: usize) {
    let n = bank.accounts.len();
    thread::scope(|s| {
        for t in 0..threads {
            s.spawn(move || {
                for r in 0..rounds {
                    let from = (t + r) % n;
                    let to = (t + r * 3 + 1) % n;
                    let _ = bank.transfer(from, to, 1 + (r % 5) as i64);
                }
            });
        }
        // An auditor reading concurrently through the RwLock.
        s.spawn(|| {
            for _ in 0..10 {
                let _ = bank.log_len();
            }
        });
    });
}

fn main() {
    let bank = Bank::new(&[1_000, 1_000, 1_000, 1_000]);
    let before = bank.total();
    simulate(&bank, 8, 1_000);
    println!("total before = {before}, after = {}", bank.total());
    println!("transfers logged = {}", bank.log_len());
    for i in 0..4 {
        println!("account {i}: {}", bank.balance(i));
    }
    println!("{:?}", bank.transfer(0, 0, 1));
    println!("{:?}", bank.transfer(0, 1, 1_000_000));

    // Poisoning demo: a thread panics while holding the lock.
    let shared = Arc::new(Mutex::new(42));
    let s2 = Arc::clone(&shared);
    let _ = thread::spawn(move || {
        let _guard = s2.lock().unwrap();
        panic!("simulated crash while holding the lock");
    })
    .join();
    println!("poisoned = {}", shared.is_poisoned());
    println!("recovered value = {}", *lock_recover(&shared));
}
