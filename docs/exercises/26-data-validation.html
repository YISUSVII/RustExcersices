Exercise 26 · Level 8 — Error Handling

Signup Form Validation

Validate raw user input into a trusted type, reporting either the first error or every error.

Intermediate"parsedon't validate"error enums with data`impl Display``?`collecting multiple errors

Instructions

Raw input arrives as SignupForm<'a> { username: &'a str, email: &'a str, age: &'a str, password: &'a str }.

  1. Implement one validator per field, each returning Result<_, ValidationError>: - validate_username(&str) -> Result<String, ValidationError> — trimmed, non-empty (EmptyUsername), at most 16 chars (UsernameTooLong { max: 16 }), only alphanumerics or _ (InvalidUsernameChar(char)). - validate_email(&str) -> Result<String, ValidationError> — exactly one @ with non-empty local part and a domain containing a . (InvalidEmail(String)). - validate_age(&str) -> Result<u8, ValidationError> — must parse (AgeNotANumber(String)) and be in 13..=120 (AgeOutOfRange(u32)). - validate_password(&str) -> Result<(), ValidationError> — at least 8 chars, one digit, one uppercase letter (WeakPassword(&'static str) with the reason).
  2. Implement std::fmt::Display for ValidationError with human-readable messages.
  3. Implement validate(form: &SignupForm) -> Result<ValidUser, ValidationError> — stops at the first error using ?.
  4. Implement validate_all(form: &SignupForm) -> Result<ValidUser, Vec<ValidationError>> — runs every validator and returns all errors at once.

Expected behavior

With the sample data in main:

valid: ValidUser { username: "ferris", email: "ferris@rust-lang.org", age: 30 }
first error: email 'nope' is invalid
all errors:
  - username contains invalid character '!'
  - email 'nope' is invalid
  - age must be a number, got 'ten'
  - password is too weak: needs at least 8 characters

Optional hints

  • Validate in validate with let username = validate_username(form.username)?; etc.
  • In validate_all, match on each result and push errors into a Vec.
  • Parse to u32 first so "500" produces AgeOutOfRange(500) instead of a parse error.

Bonus challenge

Implement std::error::Error for ValidationError and return Box<dyn Error> from a helper.

use std::error::Error;
use std::fmt;

#[derive(Debug)]
struct SignupForm<'a> {
    username: &'a str,
    email: &'a str,
    age: &'a str,
    password: &'a str,
}

#[derive(Debug, PartialEq)]
struct ValidUser {
    username: String,
    email: String,
    age: u8,
}

#[derive(Debug, PartialEq)]
enum ValidationError {
    EmptyUsername,
    UsernameTooLong { max: usize },
    InvalidUsernameChar(char),
    InvalidEmail(String),
    AgeNotANumber(String),
    AgeOutOfRange(u32),
    WeakPassword(&'static str),
}

impl fmt::Display for ValidationError {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        match self {
            ValidationError::EmptyUsername => write!(f, "username must not be empty"),
            ValidationError::UsernameTooLong { max } => {
                write!(f, "username must be at most {max} characters")
            }
            ValidationError::InvalidUsernameChar(c) => {
                write!(f, "username contains invalid character '{c}'")
            }
            ValidationError::InvalidEmail(email) => write!(f, "email '{email}' is invalid"),
            ValidationError::AgeNotANumber(raw) => write!(f, "age must be a number, got '{raw}'"),
            ValidationError::AgeOutOfRange(age) => write!(f, "age {age} is outside 13..=120"),
            ValidationError::WeakPassword(reason) => write!(f, "password is too weak: {reason}"),
        }
    }
}

impl Error for ValidationError {}

const MAX_USERNAME: usize = 16;

fn validate_username(raw: &str) -> Result<String, ValidationError> {
    let name = raw.trim();
    if name.is_empty() {
        return Err(ValidationError::EmptyUsername);
    }
    if name.chars().count() > MAX_USERNAME {
        return Err(ValidationError::UsernameTooLong { max: MAX_USERNAME });
    }
    if let Some(bad) = name.chars().find(|c| !(c.is_alphanumeric() || *c == '_')) {
        return Err(ValidationError::InvalidUsernameChar(bad));
    }
    Ok(name.to_string())
}

fn validate_email(raw: &str) -> Result<String, ValidationError> {
    let email = raw.trim();
    let invalid = || ValidationError::InvalidEmail(email.to_string());
    let (local, domain) = email.split_once('@').ok_or_else(invalid)?;
    let domain_ok = domain.contains('.') && !domain.starts_with('.') && !domain.ends_with('.');
    if local.is_empty() || domain.contains('@') || !domain_ok {
        return Err(invalid());
    }
    Ok(email.to_string())
}

fn validate_age(raw: &str) -> Result<u8, ValidationError> {
    let age: u32 = raw
        .trim()
        .parse()
        .map_err(|_| ValidationError::AgeNotANumber(raw.to_string()))?;
    match age {
        13..=120 => Ok(age as u8),
        _ => Err(ValidationError::AgeOutOfRange(age)),
    }
}

fn validate_password(raw: &str) -> Result<(), ValidationError> {
    if raw.chars().count() < 8 {
        Err(ValidationError::WeakPassword("needs at least 8 characters"))
    } else if !raw.chars().any(|c| c.is_ascii_digit()) {
        Err(ValidationError::WeakPassword("needs a digit"))
    } else if !raw.chars().any(char::is_uppercase) {
        Err(ValidationError::WeakPassword("needs an uppercase letter"))
    } else {
        Ok(())
    }
}

fn validate(form: &SignupForm) -> Result<ValidUser, ValidationError> {
    let username = validate_username(form.username)?;
    let email = validate_email(form.email)?;
    let age = validate_age(form.age)?;
    validate_password(form.password)?;
    Ok(ValidUser {
        username,
        email,
        age,
    })
}

fn validate_all(form: &SignupForm) -> Result<ValidUser, Vec<ValidationError>> {
    let username = validate_username(form.username);
    let email = validate_email(form.email);
    let age = validate_age(form.age);
    let password = validate_password(form.password);

    match (username, email, age, password) {
        (Ok(username), Ok(email), Ok(age), Ok(())) => Ok(ValidUser {
            username,
            email,
            age,
        }),
        (username, email, age, password) => {
            let errors = [username.err(), email.err(), age.err(), password.err()]
                .into_iter()
                .flatten()
                .collect();
            Err(errors)
        }
    }
}

// Bonus: ValidationError works as a boxed trait object thanks to `impl Error`.
fn validate_boxed(form: &SignupForm) -> Result<ValidUser, Box<dyn Error>> {
    Ok(validate(form)?)
}

fn main() {
    let good = SignupForm {
        username: "ferris",
        email: "ferris@rust-lang.org",
        age: "30",
        password: "Crab1234",
    };
    let bad_email = SignupForm {
        email: "nope",
        ..good
    };
    let all_bad = SignupForm {
        username: "ferris!",
        email: "nope",
        age: "ten",
        password: "short",
    };

    match validate(&good) {
        Ok(user) => println!("valid: {user:?}"),
        Err(e) => println!("unexpected error: {e}"),
    }
    if let Err(e) = validate_boxed(&bad_email) {
        println!("first error: {e}");
    }
    if let Err(errors) = validate_all(&all_bad) {
        println!("all errors:");
        for e in errors {
            println!("  - {e}");
        }
    }
}
