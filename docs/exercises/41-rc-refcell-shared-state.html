Exercise 41 · Level 13 — Smart Pointers

Rc + RefCell Shared State

Model a joint bank account shared by several owners, and build a custom smart pointer with Deref and Drop.

Intermediate`Rc<T>``RefCell<T>`interior mutability`Rc::strong_count``Rc::ptr_eq``Deref`/`DerefMut``Drop`

Instructions

  1. Define struct Account { owner: String, balance: i64 } and type SharedAccount = Rc<RefCell<Account>>;.
  2. Implement fn new_account(owner: &str, balance: i64) -> SharedAccount.
  3. Define struct Member { name: String, account: SharedAccount } and implement: - fn deposit(&self, amount: i64) - fn withdraw(&self, amount: i64) -> Result<(), String> (error if the balance is insufficient)
  4. Implement fn transfer(from: &SharedAccount, to: &SharedAccount, amount: i64) -> Result<(), String>. Reject transfers to the same account (use Rc::ptr_eq) and overdrafts.
  5. Build a custom smart pointer Tracked<T: Debug> { label, value, drop_log: Rc<RefCell<Vec<String>>> } with: - fn new(label: &'static str, value: T, drop_log: Rc<RefCell<Vec<String>>>) -> Self - impl Deref / impl DerefMut targeting T - impl Drop that pushes "dropping {label} = {value:?}" to the log.

Expected behavior

  • Two Members sharing an account see each other's deposits; Rc::strong_count is 3 (original + 2 members).
  • transfer(&a, &a, 1) → Err(..).
  • A Tracked<Vec<i32>> can call push/len directly thanks to Deref, and logs dropping numbers = [1, 2, 3] when it goes out of scope.

Optional hints

  • RefCell::borrow_mut panics if the value is already borrowed — keep borrows short and never hold two borrow_mut() on the same RefCell (that is why ptr_eq matters in transfer).
  • Drop::drop runs automatically at the end of scope; you never call it directly (use std::mem::drop(x) to drop early).

Bonus challenge

Use RefCell::try_borrow_mut in transfer to return an error instead of panicking if the account is already borrowed.

use std::cell::RefCell;
use std::ops::{Deref, DerefMut};
use std::rc::Rc;

#[derive(Debug)]
struct Account {
    owner: String,
    balance: i64,
}

type SharedAccount = Rc<RefCell<Account>>;

fn new_account(owner: &str, balance: i64) -> SharedAccount {
    Rc::new(RefCell::new(Account {
        owner: owner.to_string(),
        balance,
    }))
}

/// A household member holding a handle to a shared (joint) account.
struct Member {
    name: String,
    account: SharedAccount,
}

impl Member {
    fn deposit(&self, amount: i64) {
        self.account.borrow_mut().balance += amount;
    }

    fn withdraw(&self, amount: i64) -> Result<(), String> {
        let mut acc = self.account.borrow_mut();
        if acc.balance < amount {
            return Err(format!(
                "{} cannot withdraw {amount}: balance is {}",
                self.name, acc.balance
            ));
        }
        acc.balance -= amount;
        Ok(())
    }
}

fn transfer(from: &SharedAccount, to: &SharedAccount, amount: i64) -> Result<(), String> {
    if Rc::ptr_eq(from, to) {
        return Err("cannot transfer to the same account".to_string());
    }
    let mut src = from.borrow_mut();
    if src.balance < amount {
        return Err(format!("insufficient funds in {}'s account", src.owner));
    }
    src.balance -= amount;
    to.borrow_mut().balance += amount;
    Ok(())
}

/// A custom smart pointer that logs when it is dropped.
struct Tracked<T: std::fmt::Debug> {
    label: &'static str,
    value: T,
    drop_log: Rc<RefCell<Vec<String>>>,
}

impl<T: std::fmt::Debug> Tracked<T> {
    fn new(label: &'static str, value: T, drop_log: Rc<RefCell<Vec<String>>>) -> Self {
        Tracked {
            label,
            value,
            drop_log,
        }
    }
}

impl<T: std::fmt::Debug> Deref for Tracked<T> {
    type Target = T;
    fn deref(&self) -> &T {
        &self.value
    }
}

impl<T: std::fmt::Debug> DerefMut for Tracked<T> {
    fn deref_mut(&mut self) -> &mut T {
        &mut self.value
    }
}

impl<T: std::fmt::Debug> Drop for Tracked<T> {
    fn drop(&mut self) {
        let msg = format!("dropping {} = {:?}", self.label, self.value);
        println!("{msg}");
        self.drop_log.borrow_mut().push(msg);
    }
}

fn main() {
    let joint = new_account("Ana & Ben", 100);
    let ana = Member {
        name: "Ana".into(),
        account: Rc::clone(&joint),
    };
    let ben = Member {
        name: "Ben".into(),
        account: Rc::clone(&joint),
    };
    println!("strong_count(joint) = {}", Rc::strong_count(&joint));

    ana.deposit(50);
    ben.withdraw(30).unwrap();
    println!("joint after ops: {:?}", joint.borrow());

    println!("drop log: {:?}", log.borrow());
}
